cmake_minimum_required(VERSION 3.15)
project(GraphitAntlrRuntimeTest LANGUAGES CXX)
find_package(antlr4-runtime 4.13.2 EXACT CONFIG REQUIRED)
add_executable(runtime_version runtime_version.cpp)
target_compile_features(runtime_version PRIVATE cxx_std_17)
set_target_properties(runtime_version PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(runtime_version PRIVATE "${ANTLR4_INCLUDE_DIR}")
target_link_libraries(runtime_version PRIVATE antlr4_static)
if(WIN32)
  target_compile_definitions(runtime_version PRIVATE ANTLR4CPP_STATIC)
endif()
enable_testing()
add_test(NAME runtime_version COMMAND runtime_version)
